\procedure update-file()
\procedure notify()
<$parameters message>
<$action-createtiddler $basetitle="$:/temp/notify" text=<<message>> >
<$action-sendmessage $message="tm-notify" $param=<<createTiddler-title>>/>
</$action-createtiddler>
</$parameters>
\end notify
\procedure handle-sha()
\procedure finally()
    <!-- https://docs.github.com/en/rest/repos/contents?apiVersion=2026-03-10#create-or-update-file-contents -->
    <%if [<status>match[200]] %>
    <<notify message="Success! File saved to GitHub." >>
    <<notify message=`html_url = ${[<data>jsonget[content],[html_url]]}$` >>
    <%elseif [<status>match[201]] %>
    <<notify message="Success! File created and saved to GitHub." >>
    <%elseif [<status>match[404]] %>
    <<notify message="Error: Resource not found." >>
    <%elseif [<status>match[409]] %>
    <<notify message="Error: Conflict" >>
    <%elseif [<status>match[422]] %>
    <<notify message="Error: Validation failed, or the endpoint has been spammed." >>
    <%else%>
    <<notify message=`Error: $(status)$`  >>
    <%endif%>
\end finally
<%if [<status>match[200]] %>
<$action-sendmessage
    $message="tm-http-request"
    url=<<url>>
    method="PUT"
    basic-auth-username=<<username>>
    basic-auth-password-from-store="gh"
    header-Content-Type="application/json"
    body=`{"message": "$(commitMessage)$", "content": "$(content)$", "sha": "${ [<data>jsonget[sha]] }$"}`
    var-notify=<<notify>>
    oncompletion=<<finally>>
/>
<%else%>
<$action-sendmessage
    $message="tm-http-request"
    url=<<url>>
    method="PUT"
    basic-auth-username=<<username>>
    basic-auth-password-from-store="gh"
    header-Content-Type="application/json"
    body=`{"message": "$(commitMessage)$", "content": "$(content)$"}`
    var-notify=<<notify>>
    oncompletion=<<finally>>
/>
<%endif%>
\end handle-sha

<!-- Step 1: fetch the SHA of the RSS file -->
<$let 
    username={{!!username}}
    repo={{!!repo}}
    filePath={{!!file-path}}
    content={{{ [{!!tiddler-content}get[text]encodebase64[]]  }}}
    commitMessage=`Updated $(now)$`
    url=`https://api.github.com/repos/$(username)$/$(repo)$/contents/$(filePath)$`
>
<$action-sendmessage
    $message="tm-http-request"

    oncompletion=<<handle-sha>>

    var-repo=<<repo>>
    var-filePath=<<filePath>>
    var-content=<<content>>
    var-commitMessage=<<commitMessage>>
    var-url=<<url>>
/>
</$let>
\end update-file
